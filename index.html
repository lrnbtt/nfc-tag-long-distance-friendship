// Backend: Google Sheet + Firebase Cloud Messaging. Paste this over Code.gs. // Script property needed: SERVICE_ACCOUNT = the full contents of your Firebase service account JSON. const MOODS = { peace: '🌿 Peace', missing: '🌙 Missing you', happy: '☀️ Happy', energy: '⚡ Need energy', chaos: '🔥 Chaos' }; const sheet_ = n => SpreadsheetApp.getActive().getSheetByName(n); const json_ = o => ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); // People tab columns: token | name | group | push | birthday (push fills itself in; birthday is MM-DD, e.g. 03-15, optional) const bd_ = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'MM-dd') : String(v || '').trim(); function people_() { return sheet_('People').getDataRange().getValues().slice(1) .map((r, i) => ({ row: i + 2, token: String(r[0]), name: r[1], group: String(r[2]), push: String(r[3] || ''), birthday: bd_(r[4]) })) .filter(p => p.token); } const who_ = token => people_().find(p => p.token === String(token)); // Events tab columns: ts | group | sender | mood function doGet(e) { const me = who_(e.parameter.t); if (!me) return json_({ error: 'unknown charm' }); const events = sheet_('Events').getDataRange().getValues().slice(1) .filter(r => String(r[1]) === me.group) .map((r, i) => ({ n: i + 1, ts: new Date(r[0]).getTime(), sender: r[2], mood: r[3] })); const people = people_().filter(p => p.group === me.group).map(p => ({ name: p.name, birthday: p.birthday })); return json_({ name: me.name, events, people }); } function doPost(e) { let b; try { b = JSON.parse(e.postData.contents); } catch (err) { return json_({ error: 'bad request' }); } const me = who_(b.t); if (!me) return json_({ error: 'bad request' }); if (b.action === 'register' && b.device) { // a phone allowed buzzes sheet_('People').getRange(me.row, 4).setValue(String(b.device)); return json_({ ok: true }); } if (!MOODS[b.mood]) return json_({ error: 'bad request' }); sheet_('Events').appendRow([new Date(), me.group, me.name, b.mood]); people_().filter(p => p.group === me.group && p.token !== me.token && p.push) .forEach(p => push_(p, 'Thinking of you', me.name + ' thought of you ' + MOODS[b.mood])); return json_({ ok: true }); } function push_(p, title, body) { const proj = JSON.parse(PropertiesService.getScriptProperties().getProperty('SERVICE_ACCOUNT')).project_id; const r = UrlFetchApp.fetch('https://fcm.googleapis.com/v1/projects/' + proj + '/messages:send', { method: 'post', contentType: 'application/json', headers: { Authorization: 'Bearer ' + accessToken_() }, payload: JSON.stringify({ message: { token: p.push, data: { title: title, body: body }, webpush: { headers: { Urgency: 'high', TTL: '86400' } } } }), muteHttpExceptions: true }); if (r.getResponseCode() === 404) sheet_('People').getRange(p.row, 4).clear(); // device no longer exists } // Signs a service-account login with the private key to get a short-lived Google access token. function accessToken_() { const cache = CacheService.getScriptCache(); const hit = cache.get('fcm'); if (hit) return hit; const sa = JSON.parse(PropertiesService.getScriptProperties().getProperty('SERVICE_ACCOUNT')); const now = Math.floor(Date.now() / 1000); const b64 = x => Utilities.base64EncodeWebSafe(x).replace(/=+$/, ''); const head = b64(JSON.stringify({ alg: 'RS256', typ: 'JWT' })); const claim = b64(JSON.stringify({ iss: sa.client_email, scope: 'https://www.googleapis.com/auth/firebase.messaging', aud: 'https://oauth2.googleapis.com/token', iat: now, exp: now + 3600 })); const sig = b64(Utilities.computeRsaSha256Signature(head + '.' + claim, sa.private_key)); const res = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', { method: 'post', payload: { grant_type: 'urn:ietf:params:oauth:grant-type:jwt-bearer', assertion: head + '.' + claim + '.' + sig } }); const token = JSON.parse(res.getContentText()).access_token; cache.put('fcm', token, 3000); return token; }

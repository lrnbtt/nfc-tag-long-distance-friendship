// Backend: Google Sheet + Firebase Cloud Messaging. Paste this over Code.gs. // Script property needed: SERVICE_ACCOUNT = the full contents of your Firebase service account JSON. const LANG = 'en'; // use 'pt' in the Brazilian Portuguese sheet const STR = { en: { moods: { peace: '🌿 Peace', missing: '🌙 Missing you', happy: '☀️ Happy', energy: '⚡ Need energy', chaos: '🔥 Chaos' }, title: 'Thinking of you', body: (name, mood) => name + ' thought of you ' + mood, remind: 'Send your thought of the day 💜' }, pt: { moods: { peace: '🌿 Paz', missing: '🌙 Saudade', happy: '☀️ Feliz', energy: '⚡ Preciso de energia', chaos: '🔥 Caos' }, title: 'Pensando em você', body: (name, mood) => name + ' pensou em você ' + mood, remind: 'Mande seu pensamento do dia 💜' } }; const MOODS = STR[LANG].moods; const sheet_ = n => SpreadsheetApp.getActive().getSheetByName(n); const json_ = o => ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); // People tab columns: token | name | group | push | birthday | timezone | reminder // One row per person per friendship. Someone in several friendships uses the SAME token on each of their rows. // push fills itself in. birthday is MM-DD (03-15), optional. // timezone is a name like America/Sao_Paulo (blank = this sheet's timezone). // reminder is the local hour 0-23 for the daily reminder (blank = 19, "off" = none). const tz_ = v => { v = String(v || '').trim(); if (!v) return Session.getScriptTimeZone(); try { Utilities.formatDate(new Date(), v, 'H'); return v; } catch (e) { return Session.getScriptTimeZone(); } }; const hour_ = v => { v = String(v).trim().toLowerCase(); return v === '' ? 19 : v === 'off' ? -1 : (isFinite(v) && +v >= 0 && +v <= 23 ? Math.floor(+v) : 19); }; const day_ = (d, tz) => Utilities.formatDate(new Date(d), tz, 'yyyy-MM-dd'); const bd_ = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'MM-dd') : String(v || '').trim(); function people_() { return sheet_('People').getDataRange().getValues().slice(1) .map((r, i) => ({ row: i + 2, token: String(r[0]), name: r[1], group: String(r[2]), push: String(r[3] || ''), birthday: bd_(r[4]), tz: tz_(r[5]), hour: hour_(r[6]) })) .filter(p => p.token); } const who_ = token => people_().find(p => p.token === String(token)); const mine_ = token => people_().filter(p => p.token === String(token)); // all of one person's friendships const pick_ = (ms, g) => ms.find(m => m.group === String(g || '')) || ms[0]; // Events tab columns: ts | group | sender | mood function doGet(e) { const all = mine_(e.parameter.t); if (!all.length) return json_({ error: 'unknown charm' }); const me = pick_(all, e.parameter.g); const rows = people_(); const groups = all.map(m => ({ group: m.group, sent: !!sentMood_(m), label: rows.filter(p => p.group === m.group && p.token !== m.token).map(p => p.name).join(', ') || m.group })); const events = sheet_('Events').getDataRange().getValues().slice(1) .filter(r => String(r[1]) === me.group) .map((r, i) => ({ n: i + 1, ts: new Date(r[0]).getTime(), sender: r[2], mood: r[3] })); const people = people_().filter(p => p.group === me.group).map(p => ({ name: p.name, birthday: p.birthday })); const sm = sentMood_(me); return json_({ name: me.name, events, people, sentToday: !!sm, sentMood: sm, group: me.group, groups }); } function doPost(e) { let b; try { b = JSON.parse(e.postData.contents); } catch (err) { return json_({ error: 'bad request' }); } const all = mine_(b.t); if (!all.length) return json_({ error: 'bad request' }); const me = pick_(all, b.g); if (b.mood && all.length > 1 && me.group !== String(b.g)) return json_({ error: 'bad request' }); if (b.action === 'register' && b.device) { // a phone allowed buzzes all.forEach(m => sheet_('People').getRange(m.row, 4).setValue(String(b.device))); return json_({ ok: true }); } if (!MOODS[b.mood]) return json_({ error: 'bad request' }); const lock = LockService.getScriptLock(); // one thought per day, even on a double tap lock.waitLock(10000); try { if (sentMood_(me)) return json_({ ok: false, limit: true }); sheet_('Events').appendRow([new Date(), me.group, me.name, b.mood]); } finally { lock.releaseLock(); } people_().filter(p => p.group === me.group && p.token !== me.token && p.push) .forEach(p => push_(p, STR[LANG].title, STR[LANG].body(me.name, MOODS[b.mood]))); return json_({ ok: true }); } function push_(p, title, body, kind) { const proj = JSON.parse(PropertiesService.getScriptProperties().getProperty('SERVICE_ACCOUNT')).project_id; const r = UrlFetchApp.fetch('https://fcm.googleapis.com/v1/projects/' + proj + '/messages:send', { method: 'post', contentType: 'application/json', headers: { Authorization: 'Bearer ' + accessToken_() }, payload: JSON.stringify({ message: { token: p.push, data: { title: title, body: body, kind: kind || 'thought' }, webpush: { headers: { Urgency: 'high', TTL: '86400' } } } }), muteHttpExceptions: true }); if (r.getResponseCode() === 404) sheet_('People').getRange(p.row, 4).clear(); // device no longer exists } // Signs a service-account login with the private key to get a short-lived Google access token. function accessToken_() { const cache = CacheService.getScriptCache(); const hit = cache.get('fcm'); if (hit) return hit; const sa = JSON.parse(PropertiesService.getScriptProperties().getProperty('SERVICE_ACCOUNT')); const now = Math.floor(Date.now() / 1000); const b64 = x => Utilities.base64EncodeWebSafe(x).replace(/=+$/, ''); const head = b64(JSON.stringify({ alg: 'RS256', typ: 'JWT' })); const claim = b64(JSON.stringify({ iss: sa.client_email, scope: 'https://www.googleapis.com/auth/firebase.messaging', aud: 'https://oauth2.googleapis.com/token', iat: now, exp: now + 3600 })); const sig = b64(Utilities.computeRsaSha256Signature(head + '.' + claim, sa.private_key)); const res = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', { method: 'post', payload: { grant_type: 'urn:ietf:params:oauth:grant-type:jwt-bearer', assertion: head + '.' + claim + '.' + sig } }); const token = JSON.parse(res.getContentText()).access_token; cache.put('fcm', token, 3000); return token; } // The mood this person already sent today (in their own timezone), or '' if none yet. function sentMood_(me) { const today = day_(new Date(), me.tz); const rows = sheet_('Events').getDataRange().getValues().slice(1); for (let i = rows.length - 1; i >= 0; i--) { const r = rows[i]; if (new Date() - new Date(r[0]) > 1728e5) break; // older than two days, so it can't be today if (String(r[1]) === me.group && r[2] === me.name && day_(r[0], me.tz) === today) return r[3]; } return ''; } // Run this ONCE from the editor (choose setupReminders, press Run). It sets up an hourly check. function setupReminders() { ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'dailyReminders') .forEach(t => ScriptApp.deleteTrigger(t)); ScriptApp.newTrigger('dailyReminders').timeBased().everyHours(1).create(); } // Each hour: anyone whose local reminder hour it is, with buzzes on and at least one friendship // they haven't sent to today, gets ONE nudge (even if they are in several friendships). function dailyReminders() { const cache = CacheService.getScriptCache(), now = new Date(), byToken = {}; people_().forEach(p => (byToken[p.token] = byToken[p.token] || []).push(p)); Object.values(byToken).forEach(ms => { const p = ms.find(m => m.push) || ms[0]; if (!p.push || p.hour < 0) return; if (Number(Utilities.formatDate(now, p.tz, 'H')) !== p.hour) return; const key = 'rem:' + p.token + ':' + day_(now, p.tz); if (cache.get(key) || ms.every(m => sentMood_(m))) return; cache.put(key, '1', 7200); push_(p, STR[LANG].title, STR[LANG].remind, 'remind'); }); }
